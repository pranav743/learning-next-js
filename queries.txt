# MongoDB CRUD Queries for Order Management

# 1. Insert the sample data into the 'orders' collection
mongoimport --db order_management --collection orders --file Orders.json --jsonArray

# 2. Retrieve all orders with the order_status "shipped"
db.orders.find({ order_status: "shipped" })

# 3. Update the total_amount of the order with order_id: 1 to 70000
db.orders.updateOne({ order_id: 1 }, { $set: { total_amount: 70000 } })

# 4. Delete the order with order_id: 4
db.orders.deleteOne({ order_id: 4 })

# 5. Retrieve the order with the customer_name: "Alice Johnson"
db.orders.find({ customer_name: "Alice Johnson" })

# 6. Update the order_status of the order with order_id: 2 to "delivered"
db.orders.updateOne({ order_id: 2 }, { $set: { order_status: "delivered" } })

# 7. Retrieve all orders with a total_amount greater than or equal to 15000
db.orders.find({ total_amount: { $gte: 15000 } })
